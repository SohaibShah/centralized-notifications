# Notification Grouping Implementation Plan

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** Collapse notifications about the same subject into expandable stacks with true counts, gated by the existing admin + per-user grouping toggles.

**Architecture:** A hot-swappable `GroupingStrategy` computes a `group_key` at ingest (persisted on the row). Grouping is a keyset-paginated `grouped=true` mode of the existing feed read (one collapsed entry per stack, with per-group aggregates) plus a `?group=<key>` member filter that reuses the flat feed path. The Vue panel renders stacks inside the existing Needs action / Earlier split, with an inline peek and a "See all" group filter reusing the muted-view banner.

**Tech Stack:** pnpm workspaces; `@notifications/core` (identity-free service + Postgres reads), `@notifications/shared` (zod + types), `@notifications/server-fastify` (routes), `@notifications/vue` (component lib), `backend` (reference host), `module-sim` (test generator). Vitest units, Playwright e2e.

## Global Constraints

- **Sameness gate:** a notification's group is a derived `group_key`; the **only** grouping signal. Priority/audience/mute are independent.
- **Default strategy precedence (verbatim):** (1) `metadata.groupKey` if a non-empty string → `${module}:${groupKey}`; else (2) first entity token in the title (`[A-Z]{2,10}-\d{1,10}` or `#\d{1,10}`) → instance key `${module}:${entityNorm}`; else (3) normalized title template → kind key `${module}:${category ?? "_"}:${template}`; else (4) `null` (standalone). Deterministic, pure.
- **Persisted at ingest**, never at read. `group_key`/`group_label` nullable; `null` = standalone.
- **Reads stay NFR-2:** keyset only, no OFFSET, no full-table total. Grouped read is `DISTINCT ON`/window + keyset.
- **Scope to the authenticated `user_key`** everywhere; `@notifications/core` stays identity-free (no `FROM users`; `boundary.test.ts` enforced).
- **Toggle gating:** grouping active iff `settings.flags.groupingEnabled` (admin) AND `preferences.groupingEnabled` (user). Either off → flat feed. Both toggles already exist; no new persistence for them.
- **`priority_rank`:** `0=critical, 1=high, 2=normal, 3=low` (ascending = more severe).
- **Conventions:** TS strict; zod at the boundary; parameterized SQL; Conventional Commits; **no AI commit trailers**; `pnpm lint` + `pnpm typecheck` clean before "done"; new business logic gets a Vitest test; UI verified via Playwright/`browser-tester`, not just `tsc`.
- **v1 scope boundaries:** filter chips/search active ⇒ flat (ungrouped); muted view stays flat; no per-stack "mark all read"; AI/similarity clustering out of scope (seam only).

- [ ] **Step 4: Implement `text-strategy.ts`**

`packages/core/src/grouping/text-strategy.ts`:

```ts
import type { Notification } from "@notifications/shared";
import type { GroupAssignment, GroupingStrategy } from "./types";

// Anchored, bounded patterns (title is <= 500 chars). Order = precedence for the earliest match.
const ENTITY_PATTERNS: RegExp[] = [
  /\b[A-Za-z]{2,10}-\d{1,10}\b/, // DSAR-1042
  /#\d{1,10}\b/, // #1042
];
// Volatile tokens stripped to form a "kind" template. All global, linear-time.
const VOLATILE: RegExp[] = [
  /\b[A-Za-z]{2,10}-\d{1,10}\b/g, // ids like DSAR-1042
  /#\d{1,10}\b/g, // #1042
  /\b\d[\d.,:/-]*\b/g, // numbers, dates, times
  /\b[0-9a-f]{8,}\b/gi, // hex / uuid-ish runs
];

/** Earliest entity match across ENTITY_PATTERNS, or null. */
function firstEntity(title: string): { text: string; end: number } | null {
  let best: { text: string; end: number } | null = null;
  for (const re of ENTITY_PATTERNS) {
    const m = re.exec(title);
    if (m && (best === null || m.index < best.end - m[0].length)) {
      if (best === null || m.index < best.end - best.text.length)
        best = { text: m[0], end: m.index + m[0].length };
    }
  }
  return best;
}

/** Lowercased title with volatile tokens and non-letters stripped, whitespace collapsed. */
function templateOf(title: string): string {
  let t = title.toLowerCase();
  for (const re of VOLATILE) t = t.replace(re, " ");
  t = t
    .replace(/[^\p{L}\s]/gu, " ")
    .replace(/\s+/g, " ")
    .trim();
  return t;
}

function capitalize(s: string): string {
  return s.length === 0 ? s : s[0].toUpperCase() + s.slice(1);
}

export function createTextGroupingStrategy(): GroupingStrategy {
  return {
    keyFor(n: Notification): GroupAssignment | null {
      // (1) explicit module-provided key
      const explicit = n.metadata?.["groupKey"];
      if (typeof explicit === "string" && explicit.trim() !== "") {
        return { key: `${n.module}:${explicit.trim()}`, label: n.title };
      }
      // (2) instance: earliest entity token in the title
      const ent = firstEntity(n.title);
      if (ent) {
        const norm = ent.text.toLowerCase().replace(/\s+/g, "");
        return { key: `${n.module}:${norm}`, label: n.title.slice(0, ent.end).trim() };
      }
      // (3) kind: normalized template
      const template = templateOf(n.title);
      if (template === "") return null;
      return {
        key: `${n.module}:${n.category ?? "_"}:${template}`,
        label: capitalize(template).slice(0, 60),
      };
    },
  };
}
```

> Note on `firstEntity`: keep it simple and correct — replace the body with a clean "scan each pattern, keep the match with the smallest `index`" loop if the above index math reads awkwardly during implementation; the tests pin the required behavior.

```ts
import type { QueryFn } from "../db";
import type { GroupingStrategy } from "../grouping/types";
import type { Notification } from "@notifications/shared";

/**
 * Re-key every notification by running `strategy.keyFor` and updating group_key/group_label.
 * Idempotent and safe to re-run. Processes in id-keyset batches so it never loads the whole table.
 * Intended for a strategy swap or the initial rollout; run from the reference host.
 */
export async function backfillGroupKeys(
  query: QueryFn,
  strategy: GroupingStrategy,
  batchSize = 500,
): Promise<{ updated: number }> {
  let after = "";
  let updated = 0;
  for (;;) {
    const { rows } = await query<{
      id: string;
      module: string;
      title: string;
      category: string | null;
      metadata: Record<string, unknown> | null;
      priority: Notification["priority"];
    }>(
      `SELECT id, module, title, category, metadata, priority
         FROM notifications WHERE id > $1 ORDER BY id LIMIT $2`,
      [after, batchSize],
    );
    if (rows.length === 0) break;
    for (const r of rows) {
      const g = strategy.keyFor({
        id: r.id,
        module: r.module,
        title: r.title,
        description: "",
        priority: r.priority,
        snoozable: true,
        audience: { scope: "global" },
        ...(r.category != null ? { category: r.category } : {}),
        ...(r.metadata != null ? { metadata: r.metadata } : {}),
      });
      await query("UPDATE notifications SET group_key = $2, group_label = $3 WHERE id = $1", [
        r.id,
        g?.key ?? null,
        g?.label ?? null,
      ]);
      updated++;
    }
    after = rows[rows.length - 1].id;
  }
  return { updated };
}
```

```ts
import { z } from "zod";
import type { GroupedPage, NotificationPriority } from "@notifications/shared";
import type { QueryFn } from "../db";
import type { Principal } from "../types";
import { audienceWhere } from "../audience/match";
import { muteWhere } from "../preferences/mute";
import { parseActions } from "./feed";

const DEFAULT_LIMIT = 25;
const MAX_LIMIT = 100;

interface GCursor {
  g: true;
  ts: string;
  id: string;
}
const gCursorSchema = z.object({
  g: z.literal(true),
  ts: z.string().datetime({ offset: true }),
  id: z.string().min(1),
});
const enc = (c: GCursor) => Buffer.from(JSON.stringify(c)).toString("base64url");
function dec(raw: string): GCursor | null {
  try {
    const p = gCursorSchema.safeParse(JSON.parse(Buffer.from(raw, "base64url").toString("utf8")));
    return p.success ? p.data : null;
  } catch {
    return null;
  }
}

export interface GroupedArgs {
  principal: Principal;
  cursor?: string;
  limit?: number;
}
export type GroupedResult =
  { ok: true; page: GroupedPage } | { ok: false; error: "invalid cursor" };

/**
 * One collapsed entry per group (its most-recent member = representative), annotated with per-group
 * total / unread / top priority via window aggregates. Standalone rows (group_key IS NULL) are their
 * own entries. Keyset-paginated by the representative (created_at, id); NFR-2 (no OFFSET, no total).
 * `entry_key` partitions by group_key, falling back to id so each standalone is its own partition.
 */
export async function listGrouped(query: QueryFn, args: GroupedArgs): Promise<GroupedResult> {
  const limit = Math.min(Math.max(args.limit ?? DEFAULT_LIMIT, 1), MAX_LIMIT);
  let cursor: GCursor | null = null;
  if (args.cursor !== undefined) {
    cursor = dec(args.cursor);
    if (!cursor) return { ok: false, error: "invalid cursor" };
  }

  const params: unknown[] = [args.principal.userKey];
  const audience = audienceWhere(args.principal, params);
  const mute = muteWhere(args.principal.userKey, params);

  let keyset = "";
  if (cursor) {
    params.push(cursor.ts, cursor.id);
    keyset = `WHERE (created_at, id) < ($${params.length - 1}::timestamptz, $${params.length}::text)`;
  }
  params.push(limit + 1);
  const limitP = `$${params.length}`;

  const { rows } = await query<{
    id: string;
    module: string;
    title: string;
    description: string;
    priority: NotificationPriority;
    snoozable: boolean;
    category: string | null;
    audience_scope: string;
    audience_id: string | null;
    actions: unknown[] | null;
    metadata: Record<string, unknown> | null;
    source_ts: Date | null;
    created_iso: string;
    read: boolean;
    group_key: string | null;
    group_label: string | null;
    group_total: number;
    group_unread: number;
    top_priority: NotificationPriority;
  }>(
    `WITH scoped AS (
       SELECT n.*, (r.user_key IS NOT NULL) AS read,
              COALESCE(n.group_key, n.id) AS entry_key
         FROM notifications n
         LEFT JOIN notification_reads r ON r.notification_id = n.id AND r.user_key = $1
        WHERE n.suppressed = false AND ${audience} AND ${mute}
     ),
     ranked AS (
       SELECT *,
         row_number() OVER (PARTITION BY entry_key ORDER BY created_at DESC, id DESC) AS rn,
         count(*)                        OVER (PARTITION BY entry_key) AS group_total,
         count(*) FILTER (WHERE read = false) OVER (PARTITION BY entry_key) AS group_unread,
         first_value(priority) OVER (PARTITION BY entry_key ORDER BY priority_rank ASC
                                     ROWS BETWEEN UNBOUNDED PRECEDING AND UNBOUNDED FOLLOWING) AS top_priority
         FROM scoped
     )
     SELECT id, module, title, description, priority, snoozable, category,
            audience_scope, audience_id, actions, metadata, source_ts,
            group_key, group_label, group_total::int AS group_total,
            group_unread::int AS group_unread, top_priority,
            to_char(created_at AT TIME ZONE 'UTC', 'YYYY-MM-DD"T"HH24:MI:SS.USZ') AS created_iso,
            read
       FROM ranked
       WHERE rn = 1 ${keyset ? "AND" + keyset.slice(5) : ""}
       ORDER BY created_at DESC, id DESC
       LIMIT ${limitP}`,
    params,
  );

  const hasMore = rows.length > limit;
  const pageRows = hasMore ? rows.slice(0, limit) : rows;
  const last = pageRows[pageRows.length - 1];
  const entries = pageRows.map((row) => ({
    id: row.id,
    module: row.module,
    title: row.title,
    description: row.description,
    priority: row.priority,
    snoozable: row.snoozable,
    ...(row.category != null ? { category: row.category } : {}),
    audience:
      row.audience_scope === "global"
        ? { scope: "global" as const }
        : {
            scope: row.audience_scope as "team" | "role" | "user",
            id: row.audience_id ?? undefined,
          },
    ...(row.actions != null ? { actions: parseActions(row.actions) } : {}),
    ...(row.metadata != null ? { metadata: row.metadata } : {}),
    ...(row.source_ts != null ? { timestamp: row.source_ts.toISOString() } : {}),
    createdAt: row.created_iso,
    read: row.read,
    ...(row.group_key != null ? { groupKey: row.group_key } : {}),
    ...(row.group_label != null ? { groupLabel: row.group_label } : {}),
    groupTotal: row.group_total,
    groupUnread: row.group_unread,
    topPriority: row.top_priority,
  }));
  return {
    ok: true,
    page: {
      entries,
      nextCursor: hasMore && last ? enc({ g: true, ts: last.created_iso, id: last.id }) : null,
    },
  };
}
```

> Implementation note: `keyset` here is applied inside the `ranked` outer query on the representative `(created_at, id)`. If the inline `keyset.slice(5)` splice reads awkwardly, build the outer WHERE explicitly as `WHERE rn = 1` plus an optional `AND (created_at, id) < (...)` — the tests pin correct keyset behavior.
